#requires -Version 5.1
#requires -RunAsAdministrator
[CmdletBinding()]
param([ValidateRange(1,2147483647)][int]$InterfaceIndex=<LOCAL_INTERFACE_INDEX>)
$ErrorActionPreference='Stop'
Set-StrictMode -Version Latest
$repo=(Resolve-Path (Join-Path $PSScriptRoot '../..')).Path
$session='WifiCancel-'+[Guid]::NewGuid().ToString('N').Substring(0,12)
$output=Join-Path $PSScriptRoot $session
$started=$false;$result=0
try {
    New-Item -ItemType Directory -Path $output -ErrorAction Stop | Out-Null
    foreach($pin in @(
        @{Name='wlanapi.dll';Hash='925f1c50e1c86d625aeb87e39a307e067a3e6f54eedbd8a598d64423203c234b'},
        @{Name='locationframework.dll';Hash='1ab04f90fdd1b0b920e4cf0f50bdde41b2824804f4006eed7a532d83bfd108e0'}
    )){if((Get-FileHash -LiteralPath (Join-Path $env:WINDIR ('System32/'+$pin.Name)) -Algorithm SHA256).Hash -ine $pin.Hash){throw 'Unqualified Windows DLL'}}
    & python (Join-Path $repo 'tools/qualcomm_probe.py') --if-index $InterfaceIndex --command tsf_read_value --output (Join-Path $output 'driver-preview.json') | Out-Null
    if($LASTEXITCODE -ne 0){throw 'Driver qualification failed'}
    [pscustomobject]@{SessionName=$session;StartedUtc=[DateTime]::UtcNow.ToString('o');CancelAttempts=1;RecoveryRequests=1} | ConvertTo-Json | Set-Content -LiteralPath (Join-Path $output 'session.json') -Encoding UTF8
    & logman start $session -p '{bb6f5b93-635c-47be-816f-e895e77064a8}' 0x2000000000000010 0xff -o (Join-Path $output 'cancel.etl') -f bincirc -max 8 -ets | Set-Content -LiteralPath (Join-Path $output 'trace-start.txt') -Encoding UTF8
    if($LASTEXITCODE -ne 0){throw 'Trace start failed'}
    $started=$true
    & (Join-Path $PSScriptRoot 'ftm_cancel_once.exe') --execute $InterfaceIndex (Join-Path $output 'cancel-response.bin') | Set-Content -LiteralPath (Join-Path $output 'cancel-result.json') -Encoding UTF8
    if($LASTEXITCODE -ne 0){throw 'Cancellation did not reach a completed callback; recovery request skipped'}
    Start-Sleep -Milliseconds 1000
    & (Join-Path $repo 'artifacts/ftm_once.exe') --execute $InterfaceIndex (Join-Path $output 'recovery-response.bin') | Set-Content -LiteralPath (Join-Path $output 'recovery-result.json') -Encoding UTF8
    if($LASTEXITCODE -ne 0){throw 'Normal recovery FTM request failed'}
    Start-Sleep -Milliseconds 1000
} catch {
    $result=1;Write-Warning $_.Exception.Message
    if(Test-Path $output){[pscustomobject]@{Error=$_.Exception.Message} | ConvertTo-Json | Set-Content -LiteralPath (Join-Path $output 'capture-error.json') -Encoding UTF8}
} finally {
    if($started){& logman stop $session -ets | Set-Content -LiteralPath (Join-Path $output 'trace-stop.txt') -Encoding UTF8;if($LASTEXITCODE -ne 0){$result=1}}
    try {$a=@(Get-NetAdapter -IncludeHidden -ErrorAction Stop | Where-Object {$_.ifIndex -eq $InterfaceIndex});if($a.Count -ne 1){throw 'Missing adapter'};$a[0]|Select-Object Name,Status,ifIndex,DriverVersion|ConvertTo-Json|Set-Content -LiteralPath (Join-Path $output 'adapter-after.json') -Encoding UTF8;if($a[0].Status -ne 'Up' -or $a[0].DriverVersion -ne '1.0.4374.1300'){$result=1}}catch{$result=1;Write-Warning $_.Exception.Message}
}
Write-Output $output
exit $result
